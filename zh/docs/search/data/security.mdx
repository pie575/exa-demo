export const PlaygroundQuery = ({query, category, filters}) => {
  const PLAYGROUND = "https://dashboard.exa.ai/playground/search";
  const DEFAULT_FILTERS = {
    type: "auto",
    highlights: true
  };
  const params = [`q=${encodeURIComponent(query)}`];
  if (category) params.push(`c=${encodeURIComponent(category)}`);
  params.push(`filters=${encodeURIComponent(JSON.stringify({
    ...DEFAULT_FILTERS,
    ...filters
  }))}`);
  const href = `${PLAYGROUND}?${params.join("&")}`;
  return <div className="playground-query not-prose">
      <code className="playground-query-text">{query}</code>
      <a className="playground-query-run" href={href} target="_blank" rel="noreferrer" title="在 API Playground 中打开" aria-label={`在 API Playground 中打开“${query}”`}>
        {}
        <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
          <path d="M21 13v6a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V5a2 2 0 0 1 2-2h6" />
          <path d="m21 3-9 9" />
          <path d="M15 3h6v6" />
        </svg>
      </a>
    </div>;
};

使用 Exa Search，从安全团队日常关注的来源中获取漏洞记录、厂商安全公告和威胁研究。

## 涵盖范围 {#included}

* CVE 与 GHSA 漏洞记录
* 厂商安全公告与补丁说明
* 威胁情报报告与安全事件复盘
* 信任页面、子处理者清单与合规文档
* 安全博客、会议演讲与研究成果

## 适用场景 {#use-it-for}

* 漏洞分级与暴露面评估
* 威胁情报与攻击者追踪
* 厂商风险与第三方安全审查
* 安全监控与告警

## query示例 {#example-queries}

### 分类处理某一类漏洞 {#triage-a-vulnerability-class}

指明产品、版本范围和严重程度。

<PlaygroundQuery query="critical CVEs affecting Apache Struts 6.x" />

### 查找厂商安全公告 {#find-vendor-advisories}

描述漏洞的利用状态和产品类别，而不是单个 CVE ID。

<PlaygroundQuery query="vendor advisories for actively exploited VPN vulnerabilities" />

### 评估厂商的安全状况 {#review-a-vendors-security-posture}

指明文档类型和厂商类别。

<PlaygroundQuery query="subprocessor lists for SOC 2 compliant CRM vendors" />

### 研究攻击方 {#research-an-adversary}

写明团伙或行动的名称，以及你关注的技术手法或行业领域。

<PlaygroundQuery query="reports on ransomware groups targeting healthcare providers this year" />

## 发起请求 {#make-a-request}

<CodeGroup>
  ```python Python theme={null}
  from exa_py import Exa

  exa = Exa()

  results = exa.search(
      "critical CVEs affecting Apache Struts 6.x",
      type="auto",
      num_results=10,
  )
  ```

  ```javascript JavaScript theme={null}
  import Exa from "exa-js";

  const exa = new Exa();

  const results = await exa.search(
    "critical CVEs affecting Apache Struts 6.x",
    {
      type: "auto",
      numResults: 10,
    }
  );
  ```

  ```bash cURL theme={null}
  curl -s -X POST https://api.exa.ai/search \
    -H "Authorization: Bearer $EXA_API_KEY" \
    -H "Content-Type: application/json" \
    -d '{
      "query": "critical CVEs affecting Apache Struts 6.x",
      "type": "auto",
      "numResults": 10
    }'
  ```
</CodeGroup>
